\hypersetup{
  colorlinks=true,
  linkcolor=purple,
  urlcolor=purple,
  citecolor=purple,
}

% ===== Line Spacing: 29.5磅 =====
% Target baseline: 29.5pt for 三号(16pt)
% ctex 三号 default baseline ≈ 19.8pt → factor = 29.5/19.8 ≈ 1.49
% Using setspace which handles ctex's \zihao baseline resets correctly
\setstretch{1.49}

% ===== Paragraph Indent: 首行缩进两字符 (set after \ccwd becomes available) =====
\AtBeginDocument{%
  \zihao{3}%       三号 (16pt)
  \fangsong%        正文: 仿宋
  \setlength{\parindent}{2\ccwd}%  首行缩进两字符
  \setlength{\parskip}{0pt}%
}

% ===== Page Number: 居中插入页码 =====
\pagestyle{fancy}
\fancyhf{}
\fancyfoot[C]{\zihao{5}---\ \thepage\ ---}
\renewcommand{\headrulewidth}{0pt}
\renewcommand{\footrulewidth}{0pt}

% ===== Heading Customization =====
% Pandoc output mapping (after markdown preprocessing shift):
%   \section        → 一级标题 "一、"     (黑体三号, 不加粗)
%   \subsection     → 二级标题 "（一）"   (楷体三号, 不加粗)
%   \subsubsection  → 三级标题 "1."       (仿宋三号, 不加粗)
\ctexset{
  section = {
    numbering = false,
    format = \zihao{3}\heiti,
    indent = 0pt,
    beforeskip = 0.8\baselineskip,
    afterskip = 0.4\baselineskip,
  },
  subsection = {
    numbering = false,
    format = \zihao{3}\kaishu,
    indent = 0pt,
    beforeskip = 0.6\baselineskip,
    afterskip = 0.3\baselineskip,
  },
  subsubsection = {
    numbering = false,
    format = \zihao{3}\fangsong,
    indent = 0pt,
    beforeskip = 0.4\baselineskip,
    afterskip = 0.2\baselineskip,
  },
}

% ===== Blockquote: 左侧紫色竖线, 正文仿宋 =====
\renewenvironment{leftbar}{%
  \def\FrameCommand{{\color{purple}\vrule width 2pt}\hspace{10pt}}%
  \MakeFramed{\advance\hsize-\width\FrameRestore}%
}{\endMakeFramed}
\renewenvironment{quote}{%
  \smallskip
  \begin{leftbar}%
}{%
  \end{leftbar}%
}

% ===== Code Block (verbatim): 等宽小五号, 紧凑行距 =====
\BeforeBeginEnvironment{verbatim}{\begingroup\zihao{-5}}
\AfterEndEnvironment{verbatim}{\endgroup}

% ===== Table: 小五号以适应宽表格 =====
% Use AtBeginEnvironment to scope the font change INSIDE the group
\AtBeginEnvironment{longtable}{\zihao{-5}\renewcommand{\arraystretch}{1.3}}
\AtBeginEnvironment{tabular}{\zihao{-5}\renewcommand{\arraystretch}{1.3}}

% ===== Image Handling: pandocbounded must be defined =====
% Without this, pandoc-generated \pandocbounded{...} silently fails
% and all images are dropped from the PDF
\newcommand{\pandocbounded}[1]{%
  \begin{center}
  \makebox[\textwidth][c]{#1}%
  \end{center}
}

% ===== Compact Lists =====
\setlist{nosep,leftmargin=2\ccwd,itemsep=0pt,topsep=0pt,partopsep=0pt}

  \egroup
}

$endif$
