% ==================== 封面 TikZ 图案（左侧 Agent 网络节点）====================
\newcommand{\covergeometry}{%
  \begin{tikzpicture}[scale=0.15]
    % 背景节点网络
    \foreach \x in {0,8,...,80} {
      \foreach \y in {0,12,...,120} {
        \pgfmathsetmacro{\s}{random(0,100)}
        \ifnum\s<18
          \fill[white!10] (\x,\y) circle (0.6);
        \fi
        \ifnum\s<6
          \fill[white!20] (\x,\y) circle (1.0);
        \fi
      }
    }
    % 连接线
    \foreach \i in {1,...,25} {
      \pgfmathsetmacro{\xa}{random(0,80)}
      \pgfmathsetmacro{\ya}{random(0,120)}
      \pgfmathsetmacro{\xb}{\xa + random(-20,20)}
      \pgfmathsetmacro{\yb}{\ya + random(-25,25)}
      \draw[white!8, line width=0.3] (\xa,\ya) -- (\xb,\yb);
    }
    % 亮点节点
    \foreach \i in {1,...,8} {
      \pgfmathsetmacro{\x}{random(10,70)}
      \pgfmathsetmacro{\y}{random(15,105)}
      \fill[white!30] (\x,\y) circle (2.0);
      \draw[white!15, line width=0.5] (\x,\y) circle (3.5);
    }
  \end{tikzpicture}%
}
